4//! Transport-level plumbing shared by the CTA and EOS clients.
5//!
6//! The central type is [`EndpointConfig`]: it describes *where* to connect
7//! (a `http://` or `https://` [`Url`]), *how* to authenticate ([`JwtAuth`]) and
8//! which TLS trust material to use. [`EndpointConfig::build_channel`] turns that
9//! description into a connected [`Channel`], and
10//! [`AuthorizationInterceptor`] attaches the bearer token to every outgoing
11//! request.
12
13use std::{path::PathBuf, string::FromUtf8Error};
14
15use tonic::{
16    metadata::MetadataValue,
17    service::Interceptor,
18    transport::{Certificate, Channel, ClientTlsConfig},
19};
20use url::Url;
21
22use crate::errors::{Error, validate_scheme};

24/// A JSON Web Token used to authenticate against CTA and EOS.
25#[derive(Clone)]
26pub struct JwtAuth {
27    /// The token as a string
28    pub token: String,
29}
30
31impl std::fmt::Debug for JwtAuth {
32    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
33        f.debug_struct("JwtAuth")
34            .field("token", &"[redacted]")
35            .finish()
36    }
37}
38
39impl JwtAuth {
40    /// Wraps the raw bytes of a token.
41    ///
42    /// # Errors
43    ///
44    /// It can fail if the token is malformed (UTF-8 Error)
45    pub fn new(token: Vec<u8>) -> Result<Self, FromUtf8Error> {
46        Ok(Self {
47            token: String::from_utf8(token)?,
48        })
49    }
50}

52/// Everything needed to open an authenticated gRPC channel to a service.
53#[derive(Debug, Clone)]
54pub struct EndpointConfig {
55    /// Service endpoint (`http://` or `https://`)
56    pub endpoint: Url,
57    /// Credentials presented to the service (only JWT supported for now)
58    pub authentication: JwtAuth,
59    /// Optional PEM bundle with additional CA certificates to trust. When
60    /// unset, the platform trust store is used.
61    pub ca_cert_bundle: Option<PathBuf>,
62    /// Optional hostname to validate the server certificate against, for cases
63    /// where the connection is made to an address that does not match the
64    /// certificate's subject (e.g. a port-forwarded development cluster).
65    pub alternative_cta_hostname: Option<String>,
66}
67
68impl EndpointConfig {
69    /// Assembles a configuration from its parts.
70    ///
71    /// See the field documentation of [`EndpointConfig`] for the meaning of the
72    /// optional arguments.
73    pub fn new(
74        endpoint: Url,
75        authentication: JwtAuth,
76        ca_cert_bundle: Option<PathBuf>,
77        alternative_cta_hostname: Option<String>,
78    ) -> Self {
79        Self {
80            endpoint,
81            authentication,
82            ca_cert_bundle,
83            alternative_cta_hostname,
84        }
85    }
86
87    /// Connects to [`Self::endpoint`] and returns the resulting channel.
88    ///
89    /// # Errors
90    ///
91    /// Returns [`Error::UnsupportedScheme`] if the endpoint scheme is not one
92    /// of `SUPPORTED_SCHEMES`, [`Error::InvalidURI`] if the endpoint cannot
93    /// be used as a gRPC URI, [`Error::IO`] if the CA bundle cannot be read and
94    /// [`Error::Transport`] if TLS setup or the connection itself fails.
95    pub async fn build_channel(&self) -> Result<Channel, Error> {
96        // Only `http` is plaintext; every other supported scheme gets TLS.
97        validate_scheme(&self.endpoint)?;
98        let insecure = self.endpoint.scheme() == "http";
99
100        let endpoint = Channel::from_shared(self.endpoint.to_string())
101            .map_err(|_| Error::InvalidURI(self.endpoint.clone().into()))?;
102
103        let endpoint = if insecure {
104            endpoint
105        } else {
106            let mut tls: ClientTlsConfig = ClientTlsConfig::new();
107
108            if let Some(file_path) = &self.ca_cert_bundle {
109                let data = tokio::fs::read(file_path).await.map_err(Error::IO)?;
110                tls = tls.ca_certificate(Certificate::from_pem(data));
111            }
112
113            if let Some(hostname) = &self.alternative_cta_hostname {
114                tls = tls.domain_name(hostname);
115            }
116
117            endpoint.tls_config(tls).map_err(Error::Transport)?
118        };
119
120        let channel = endpoint.connect().await.map_err(Error::Transport)?;
121
122        Ok(channel)
123    }
124}

126/// A [`tonic`] interceptor that adds `authorization: Bearer <token>` to every
127/// request sent through the intercepted service.
128pub struct AuthorizationInterceptor {
129    token: MetadataValue<tonic::metadata::Ascii>,
130}
131
132impl AuthorizationInterceptor {
133    /// Pre-renders the authorization header from the given credentials.
134    ///
135    /// # Errors
136    ///
137    /// Returns [`Error::InvalidMetadata`] if the token cannot be represented as an ASCII
138    /// metadata value (for example because it contains a line break).
139    pub fn new(auth: crate::rpc::JwtAuth) -> Result<Self, Error> {
140        let token: MetadataValue<_> = format!("Bearer {}", auth.token).parse()?;
141
142        Ok(Self { token })
143    }
144}
145
146impl Interceptor for AuthorizationInterceptor {
147    fn call(&mut self, mut req: tonic::Request<()>) -> Result<tonic::Request<()>, tonic::Status> {
148        log::trace!("Adding authorization: {:#?}", self.token);
149        req.metadata_mut()
150            .insert("authorization", self.token.clone());
151        Ok(req)
152    }
153}

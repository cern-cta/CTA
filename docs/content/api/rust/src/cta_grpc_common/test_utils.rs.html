4// Helpers to build gRPC response streams without a server, for unit tests.
5//
6// A [`Streaming<T>`] is constructed directly from a [`http_body::Body`]
7// through [`Streaming::new_response`], and we simply hand-roll the gRPC wire
8// format:
9//
10// * every message is length-delimited by a 5-byte prefix (1 byte compression
11//   flag + 4 bytes big-endian length);
12// * the body ends with a trailer frame carrying `grpc-status: 0`, which tonic
13//   requires to consider the stream properly terminated.
14//
15// This keeps the stream adapters testable at unit-test speed, with no
16// sockets, ports or spawned servers involved.
17
18//! Util functions for GRPC testing
19
20use std::convert::Infallible;
21
22use bytes::{BufMut, Bytes, BytesMut};
23use http_body::Frame;
24use http_body_util::StreamBody;
25use tonic::{
26    Status, Streaming,
27    codec::{Codec, Decoder},
28    codegen::http::{HeaderMap, HeaderValue, StatusCode},
29};
30use tonic_prost::ProstCodec;

32/// Wraps `payload` in a gRPC length-delimited frame.
33pub fn frame(payload: &[u8]) -> Bytes {
34    let mut buf = BytesMut::with_capacity(payload.len() + 5);
35    buf.put_u8(0); // not compressed
36    buf.put_u32(payload.len() as u32); // big-endian length
37    buf.put_slice(payload);
38    buf.freeze()
39}
40
41/// Encodes `messages` into a single gRPC body.
42pub fn body_of<T: prost::Message>(messages: &[T]) -> Bytes {
43    let mut buf = BytesMut::new();
44    for message in messages {
45        buf.put_slice(&frame(&message.encode_to_vec()));
46    }
47    buf.freeze()
48}
49
50/// Wraps raw gRPC frames into a body terminated by a successful trailer, as a
51/// real server would send it.
52pub fn grpc_body(data: Bytes) -> StreamBody<tokio_stream::Iter<GrpcFrames>> {
53    let mut trailers = HeaderMap::new();
54    trailers.insert("grpc-status", HeaderValue::from_static("0"));
55
56    StreamBody::new(tokio_stream::iter(vec![
57        Ok(Frame::data(data)),
58        Ok(Frame::trailers(trailers)),
59    ]))
60}
61
62/// The frame sequence of a canned gRPC body.
63pub type GrpcFrames = std::vec::IntoIter<Result<Frame<Bytes>, Infallible>>;
64
65/// Builds a decoder for `T` using the same prost codec the generated clients use.
66pub fn decoder_for<T>() -> impl Decoder<Item = T, Error = Status> + Send + 'static
67where
68    T: prost::Message + Default + 'static,
69{
70    ProstCodec::<T, T>::default().decoder()
71}
72
73/// Builds a response stream that yields `messages`, as if a server had sent
74/// them.
75pub fn streaming_response<T>(messages: &[T]) -> Streaming<T>
76where
77    T: prost::Message + Default + 'static,
78{
79    streaming_response_from_bytes(body_of(messages))
80}
81
82/// Builds a response stream from raw frames, for tests that need malformed
83/// or hand-crafted input.
84pub fn streaming_response_from_bytes<T>(body: Bytes) -> Streaming<T>
85where
86    T: prost::Message + Default + 'static,
87{
88    Streaming::new_response(
89        decoder_for::<T>(),
90        grpc_body(body),
91        StatusCode::OK,
92        None,
93        None,
94    )
95}
96
97/// Builds a response stream whose single frame does not decode as `T`.
98pub fn corrupt_streaming_response<T>() -> Streaming<T>
99where
100    T: prost::Message + Default + 'static,
101{
102    // Field number 1 with wire type 6, which does not exist.
103    streaming_response_from_bytes(frame(&[0x0e, 0xff, 0xff]))
104}

20/// An iter which goes over a stream of response contents and produces the individual results.
21///
22/// The CTA frontend sends a header frame followed by an arbitrary number of data
23/// frames. This adapter enforces that the header arrives first, validates its
24/// success/failure status, skips framing artifacts and yields only the payloads
25/// ([`Data`]), so callers can treat an admin command like a plain stream of records.
26///
27/// A valid stream is always `[Header, Data*, Data*, ...]` where:
28/// - The header *must* be the first frame and *must* indicate success
29/// - Any data frames *must* follow the header
30/// - A second header is a protocol violation
31/// - A stream that ends without a header is an error
32#[must_use]
33pub struct CtaResponseIter<R = Streaming<StreamResponse>> {
34    pub(crate) response: R,
35    header_seen: bool,
36}
37
38impl<R> Stream for CtaResponseIter<R>
39where
40    R: BorrowMut<Streaming<StreamResponse>> + Unpin,
41{
42    type Item = Result<Data, CtaError>;
43
44    fn poll_next(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
45        let this = self.get_mut();
46        loop {
47            match Pin::new(this.response.borrow_mut()).poll_next(cx) {
48                Poll::Ready(Some(Ok(StreamResponse {
49                    contents: Some(contents),
50                }))) => match contents {
51                    Contents::Header(header) => {
52                        if this.header_seen {
53                            return Poll::Ready(Some(Err(CtaError::UnexpectedResponseType(
54                                ResponseType::RspErrUser,
55                            ))));
56                        }
57                        this.header_seen = true;
58
59                        if header.r#type() != ResponseType::RspSuccess {
60                            return Poll::Ready(Some(Err(CtaError::UnexpectedResponseType(
61                                header.r#type(),
62                            ))));
63                        }
64                    }
65                    Contents::Data(data) => {
66                        if !this.header_seen {
67                            return Poll::Ready(Some(Err(CtaError::UnexpectedResponseType(
68                                ResponseType::RspErrUser,
69                            ))));
70                        }
71                        if let Some(d) = data.data {
72                            return Poll::Ready(Some(Ok(d)));
73                        }
74                    }
75                },
76                Poll::Ready(Some(Ok(StreamResponse { contents: None }))) => {
77                    if !this.header_seen {
78                        return Poll::Ready(Some(Err(CtaError::UnexpectedResponseType(
79                            ResponseType::RspErrUser,
80                        ))));
81                    }
82                    return Poll::Ready(None);
83                }
84                Poll::Ready(Some(Err(e))) => {
85                    return Poll::Ready(Some(Err(CtaError::Rpc(e.into()))));
86                }
87                Poll::Ready(None) => {
88                    if !this.header_seen {
89                        return Poll::Ready(Some(Err(CtaError::UnexpectedResponseType(
90                            ResponseType::RspErrUser,
91                        ))));
92                    }
93                    return Poll::Ready(None);
94                }
95                Poll::Pending => return Poll::Pending,
96            }
97        }
98    }
99}

101/// Extension trait that turns a raw gRPC response stream into a higher-level
102/// [`Stream`] of individual items.
103#[must_use]
104pub trait StreamResponseExt: Sized {
105    /// Borrows the stream and wraps it in the adapter.
106    fn stream_response(&mut self) -> CtaResponseIter<&mut Self>;
107
108    /// Takes ownership of the stream
109    fn into_stream_response(self) -> CtaResponseIter<Self>;
110}
111
112impl StreamResponseExt for Streaming<StreamResponse> {
113    fn stream_response(&mut self) -> CtaResponseIter<&mut Self> {
114        CtaResponseIter {
115            response: self,
116            header_seen: false,
117        }
118    }
119
120    fn into_stream_response(self) -> CtaResponseIter<Self> {
121        CtaResponseIter {
122            response: self,
123            header_seen: false,
124        }
125    }
126}

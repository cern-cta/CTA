4//! Client for the CTA frontend's admin gRPC interface.
5//!
6//! The frontend exposes admin commands through two services: a unary one
7//! (`CtaRpc`, one aggregated [`Response`]) and a streaming one
8//! (`CtaRpcStream`, a sequence of [`StreamResponse`] frames). Both are wrapped
9//! by [`CtaGrpcClient`]
10//!
11//! ```no_run
12//! # use cta_client::client::CtaGrpcClient;
13//! # use cta_grpc_common::EndpointConfig;
14//! # async fn example(config: EndpointConfig) -> Result<(), Box<dyn std::error::Error>> {
15//! let unary = CtaGrpcClient::new_unary(&config).await?;
16//! let streaming = CtaGrpcClient::new_streaming(&config).await?;
17//! # Ok(())
18//! # }
19//! ```
20

40/// An authenticated client for the CTA frontend.
41/// Construct with [`CtaGrpcClient::new_unary`] or [`CtaGrpcClient::new_streaming`].
42pub struct CtaGrpcClient<C> {
43    inner: C,
44}
45
46/// The generated unary client
47pub type SyncClientType = CtaRpcClient<InterceptedService<Channel, AuthorizationInterceptor>>;
48/// The generated streaming client
49pub type StreamingClientType =
50    CtaRpcStreamClient<InterceptedService<Channel, AuthorizationInterceptor>>;
51
52impl CtaGrpcClient<SyncClientType> {
53    /// Connects to the unary admin service described by `config`.
54    ///
55    /// # Errors
56    ///
57    /// Propagates the connection errors of [`EndpointConfig::build_channel`]
58    /// and the token errors of [`AuthorizationInterceptor::new`].
59    pub async fn new_unary(config: &EndpointConfig) -> Result<Self, Error> {
60        let channel = config.build_channel().await?;
61        let client = CtaRpcClient::new(InterceptedService::new(
62            channel,
63            AuthorizationInterceptor::new(config.authentication.clone())?,
64        ));
65        Ok(Self { inner: client })
66    }
67
68    /// Sends an admin command and returns the single aggregated response.
69    ///
70    /// Note that a successful return value only means the call itself
71    /// succeeded: callers must still inspect `Response::type` to find out
72    /// whether CTA accepted the command.
73    ///
74    /// # Errors
75    ///
76    /// Returns [`Error::Rpc`] if the RPC fails.
77    pub async fn raw_admin_cmd(
78        &mut self,
79        cmd: AdminCmd,
80    ) -> Result<tonic::Response<Response>, Error> {
81        self.inner
82            .admin(Request {
83                request: Some(RequestType::Admincmd(Box::new(cmd))),
84            })
85            .await
86            .map_err(|e| Error::Rpc(RpcError::Status(e)))
87    }
88
89    /// Restores a single tape file copy in the CTA catalogue
90    /// (`recycletapefile restore`).
91    ///
92    /// The entry is identified by the tape volume, disk instance, archive file
93    /// id and copy number of `file`. `file.disk_file.id` is submitted as the
94    /// disk file id to attach the restored copy to, so callers that recreated
95    /// the EOS entry must update that field to the new id first.
96    ///
97    /// # Errors
98    ///
99    /// Fails if the RPC fails, or if the frontend answers with anything other
100    /// than a success response, in which case the frontend's message is used
101    /// as the error message.
102    pub async fn restore_deleted_file_copy(&mut self, file: &File) -> Result<(), Error> {
103        log::info!(
104            "Restoring file copy in CTA catalogue: {}",
105            file.disk_file.path
106        );
107
108        let cmd = admin_cmd!(
109            Recycletapefile.SubcmdRestore {
110                Vid: str => file.tape_file.vid,
111                Instance: str => file.disk_file.instance,
112                DiskFileId: str => file.disk_file.id,
113                ArchiveFileId: u64 => file.archive_file.id,
114                CopyNumber: u64 => file.tape_file.copy_nb,
115            }
116        );
117
118        let res = self.raw_admin_cmd(cmd).await?;
119        let res = res.into_inner();
120
121        log::info!("Restored file copy in CTA catalogue: {}", res.message_txt);
122
123        match res.r#type() {
124            ResponseType::RspSuccess => Ok(()),
125            ResponseType::RspInvalid
126            | ResponseType::RspErrProtobuf
127            | ResponseType::RspErrCta
128            | ResponseType::RspErrUser => Err(Error::UnexpectedResponseType(res.r#type())),
129        }
130    }
131}

133impl CtaGrpcClient<StreamingClientType> {
134    /// Connects to the streaming admin service described by `config`.
135    ///
136    /// # Errors
137    ///
138    /// Propagates the connection errors of [`EndpointConfig::build_channel`]
139    /// and the token errors of [`AuthorizationInterceptor::new`].
140    pub async fn new_streaming(config: &EndpointConfig) -> Result<Self, Error> {
141        let channel = config.build_channel().await.map_err(Error::Rpc)?;
142        let client = CtaRpcStreamClient::new(InterceptedService::new(
143            channel,
144            AuthorizationInterceptor::new(config.authentication.clone()).map_err(Error::Rpc)?,
145        ));
146        Ok(Self { inner: client })
147    }
148
149    /// Sends an admin command and returns the raw response stream.
150    ///
151    /// Prefer consuming the result through
152    /// [`StreamResponseExt::stream_response`],
153    /// which validates the stream header and yields the payload items directly.
154    ///
155    /// # Errors
156    ///
157    /// Returns [`Error::Rpc`] if the RPC fails.
158    pub async fn raw_admin_cmd(
159        &mut self,
160        cmd: AdminCmd,
161    ) -> Result<tonic::Streaming<StreamResponse>, Error> {
162        self.inner
163            .generic_admin_stream(Request {
164                request: Some(RequestType::Admincmd(Box::new(cmd))),
165            })
166            .await
167            .map_err(|e| Error::Rpc(RpcError::Status(e)))
168            .map(|r| r.into_inner())
169    }
170
171    /// Retrieves the file which matches the given `archive_file_id`
172    ///
173    /// # Errors
174    ///
175    /// Returns [`Error::UnexpectedStreamData`] if the returned data items do not match
176    /// the expectations or are not expected at all. [`Error::Rpc`]
177    /// can also be returned.
178    pub async fn get_file(&mut self, archive_file_id: u64) -> Result<File, Error> {
179        let cmd = admin_cmd!(Tapefile.SubcmdLs {
180            ArchiveFileId: u64 => archive_file_id
181        });
182
183        let mut res = self.raw_admin_cmd(cmd).await?;
184        let mut stream = res.stream_response();
185
186        match stream.next().await {
187            Some(Ok(Data::TflsItem(ls_item))) => {
188                log::info!("Retrieved tape file information: {:?}", ls_item);
189
190                // Sanity check: make sure we received exactly one result
191                // and that the stream does not contain unexpected additional items
192                // or errors after the first result.
193                match stream.next().await {
194                    Some(Ok(next)) => Err(Error::UnexpectedStreamData(Box::new(next))),
195                    Some(Err(e)) => Err(e),
196                    None => {
197                        let file = File::try_from(ls_item).map_err(
198                            |e: TapeFileLsItemConversionError| {
199                                Error::InvalidResponse(e.to_string())
200                            },
201                        )?;
202                        Ok(file)
203                    }
204                }
205            }
206            Some(Ok(data)) => Err(Error::UnexpectedStreamData(Box::new(data))),
207            Some(Err(e)) => Err(e),
208            None => Err(Error::NotFound("File not found".into())),
209        }
210    }
211
212    /// Queries the tape file recycle bin (`recycletapefile ls`).
213    ///
214    /// All filter arguments are optional and are combined by the frontend;
215    /// passing none of them lists the whole recycle bin. The matching records
216    /// are collected and returned to the caller.
217    ///
218    /// # Errors
219    ///
220    /// Fails if the connection cannot be established, if the frontend reports
221    /// an error for the command, if the stream contains an unexpected item
222    /// type, or if the protobuf response data is malformed (e.g., invalid
223    /// timestamps or checksums).
224    pub async fn list_deleted_files(
225        &mut self,
226        file_selector: FileSelector,
227    ) -> Result<impl Stream<Item = Result<File, Error>> + use<>, Error> {
228        let cmd = admin_cmd! (
229            Recycletapefile.SubcmdLs {
230                Vid: str? => file_selector.vid,
231                Instance: str? => file_selector.disk_instance,
232                ArchiveFileId: u64? => file_selector.archive_file_id,
233                CopyNumber: u64? => file_selector.copy_number,
234                FileId: str_list? => file_selector.file_ids,
235        });
236
237        // Execute command and get stream
238        let response_stream = self.raw_admin_cmd(cmd).await?;
239
240        Ok(response_stream
241            .into_stream_response()
242            .map(|item| match item {
243                Ok(Data::RtflsItem(ls_item)) => {
244                    File::try_from(ls_item).map_err(|e| Error::InvalidResponse(e.to_string()))
245                }
246                Ok(other) => Err(Error::UnexpectedStreamData(Box::new(other))),
247                Err(e) => Err(e),
248            }))
249    }
250}

259    /// An endpoint on the loopback interface where nothing is ever listening.
260    fn unreachable_config() -> EndpointConfig {
261        EndpointConfig::new(
262            Url::parse("http://127.0.0.1:1").expect("test endpoint should parse"),
263            JwtAuth::new(b"a.token.value".to_vec()).unwrap(),
264            None,
265            None,
266        )
267    }

269    #[tokio::test]
270    async fn new_unary_fails_on_an_unreachable_frontend() {
271        let error = CtaGrpcClient::new_unary(&unreachable_config())
272            .await
273            .err()
274            .expect("connecting to a closed port should fail");
275
276        assert!(
277            matches!(error, Error::Rpc(_)),

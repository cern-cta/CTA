21/// An error returned by [`hex_to_byte_array`] when the input string cannot be decoded.
22#[derive(Debug, thiserror::Error)]
23pub enum HexDecodeError {
24    /// A non-hexadecimal character was found in the input string.
25    #[error("non-hexadecimal character: '{0}'")]
26    NonHexCharacter(char),
27    /// An integer parsing error from `std`.
28    #[error("hexadecimal parsing error: {0}")]
29    ParseInt(#[from] ParseIntError),
30}
31
32/// Decodes a hexadecimal string into its bytes, most significant byte first.
33///
34/// An optional `0x`/`0X` prefix is accepted and an odd number of digits is
35/// treated as if it were left-padded with a zero, so `"0x1a2"` decodes to
36/// `[0x01, 0xa2]`. An empty string (or a bare prefix) decodes to an empty
37/// vector.
38///
39/// # Errors
40///
41/// Returns a [`HexDecodeError`] if the input contains anything other than
42/// hexadecimal digits, including non-ASCII characters.
43pub fn hex_to_byte_array(hex_string: &str) -> Result<Vec<u8>, HexDecodeError> {
44    let digits = hex_string
45        .strip_prefix("0x")
46        .or_else(|| hex_string.strip_prefix("0X"))
47        .unwrap_or(hex_string);
48
49    // Reject anything that is not an ASCII hex digit up front
50    if let Some(invalid) = digits.chars().find(|c| !c.is_ascii_hexdigit()) {
51        return Err(HexDecodeError::NonHexCharacter(invalid));
52    }
53
54    // From here on every character is one ASCII byte, so byte offsets and
55    // character offsets coincide.
56    let mut bytes = Vec::with_capacity(digits.len().div_ceil(2));
57    let mut offset = 0;
58
59    // An odd number of digits means the leading nibble stands alone.
60    if digits.len() % 2 == 1 {
61        bytes.push(u8::from_str_radix(&digits[..1], 16)?);
62        offset = 1;
63    }
64
65    while offset < digits.len() {
66        bytes.push(u8::from_str_radix(&digits[offset..offset + 2], 16)?);
67        offset += 2;
68    }
69
70    Ok(bytes)
71}

73impl TryInto<ChecksumType> for cta_protobuf::cta::common::checksum_blob::checksum::Type {
74    type Error = ChecksumTypeError<Self>;
75
76    fn try_into(self) -> Result<ChecksumType, Self::Error> {
77        Ok(match self {
78            Self::None => return Err(ChecksumTypeError::UnknownType(Self::None)),
79            Self::Adler32 => ChecksumType::Adler32,
80            Self::Crc32 => ChecksumType::Crc32,
81            Self::Crc32c => ChecksumType::Crc32c,
82            Self::Md5 => ChecksumType::Md5,
83            Self::Sha1 => ChecksumType::Sha1,
84            Self::Crc64 => ChecksumType::Crc64,
85            Self::Sha256 => ChecksumType::Sha256,
86            Self::Xxhash64 => ChecksumType::XxHash64,
87            Self::Blake3 => ChecksumType::Blake3,
88            Self::Hwh64 => ChecksumType::HwH64,
89        })
90    }
91}

93/// Types of checksums
94#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Display, Serialize)]
95#[non_exhaustive]
96pub enum ChecksumType {
97    /// `ADLER 32`
98    Adler32,
99    /// `CRC-32`
100    Crc32,
101    /// `CRC32C`
102    Crc32c,
103    /// `MD5`
104    Md5,
105    /// `SHA-1`
106    Sha1,
107    /// `CRC-64`
108    Crc64,
109    /// `SHA-256`
110    Sha256,
111    /// `xxHash64`
112    XxHash64,
113    /// `BLAKE3`
114    Blake3,
115    /// `HighwayHash64`
116    HwH64,
117}
118
119impl ChecksumType {
120    /// Convert a string to a [`ChecksumType`], according to EOS convention
121    ///
122    /// # Errors
123    /// Unknown hash function names will result in a [`ChecksumTypeError`]
124    pub fn try_from_eos(text: impl AsRef<str>) -> Result<Self, ChecksumTypeError<()>> {
125        Ok(match text.as_ref() {
126            "adler" | "adler32" => Self::Adler32,
127            "blake3" => Self::Blake3,
128            "crc32" => Self::Crc32,
129            "crc32c" => Self::Crc32c,
130            "md5" => Self::Md5,
131            "sha" => Self::Sha1,
132            "sha256" => Self::Sha256,
133            "crc64" => Self::Crc64,
134            "xxhash64" => Self::XxHash64,
135            "hwh64" => Self::HwH64,
136            s => return Err(ChecksumTypeError::Parse(s.into())),
137        })
138    }
139}

141/// An enum-based, unambiguous, representation of a checksum
142#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize)]
143pub struct Checksum {
144    /// The type of the checksum
145    pub r#type: ChecksumType,
146    /// The content, as a vector of bytes
147    pub value: Vec<u8>,
148}
149
150impl Checksum {
151    /// Create a checksum from its components
152    pub fn new(r#type: ChecksumType, value: Vec<u8>) -> Self {
153        Self { r#type, value }
154    }
155}
156
157impl TryFrom<&cta_protobuf::cta::admin::Checksum> for Checksum {
158    type Error = ChecksumTypeError<cta_protobuf::cta::common::checksum_blob::checksum::Type>;
159
160    fn try_from(src: &cta_protobuf::cta::admin::Checksum) -> Result<Self, Self::Error> {
161        Ok(Self {
162            r#type: src.r#type().try_into()?,
163            value: hex_to_byte_array(&src.value)
164                .map_err(|e| ChecksumTypeError::Parse(e.to_string()))?,
165        })
166    }
167}

251/// An archived (tape-resident) file entry.
252#[derive(Debug, Clone, Serialize)]
253pub struct ArchiveFile {
254    /// The archive file identifier
255    pub id: u64,
256    /// The storage class of the archive file (e.g., `cta_storage_class`)
257    pub storage_class: String,
258    /// The time at which the file was archived to tape
259    pub creation_time: DateTime<Utc>,
260    /// The size of the file in bytes
261    pub size: u64,
262    /// Checksums computed over the file content
263    pub checksums: Vec<Checksum>,
264}
265
266impl TryFrom<cta_protobuf::cta::admin::tape_file_ls_item::ArchiveFile> for ArchiveFile {
267    type Error = ArchiveFileConversionError;
268
269    fn try_from(
270        val: cta_protobuf::cta::admin::tape_file_ls_item::ArchiveFile,
271    ) -> Result<Self, Self::Error> {
272        let creation_time = DateTime::from_timestamp_secs(val.creation_time as i64).ok_or(
273            ArchiveFileConversionError::InvalidTimestamp(val.creation_time as i64),
274        )?;
275
276        let checksums = val
277            .checksum
278            .iter()
279            .map(Checksum::try_from)
280            .collect::<Result<Vec<_>, _>>()
281            .map_err(|e| ArchiveFileConversionError::ChecksumConversion(e.to_string()))?;
282
283        Ok(ArchiveFile {
284            id: val.archive_id,
285            storage_class: val.storage_class,
286            creation_time,
287            size: val.size,
288            checksums,
289        })
290    }
291}

306impl TryFrom<RecycleTapeFileLsItem> for File {
307    type Error = RecycleTapeFileConversionError;
308
309    fn try_from(val: RecycleTapeFileLsItem) -> Result<Self, Self::Error> {
310        let creation_time = DateTime::from_timestamp_secs(val.archive_file_creation_time as i64)
311            .ok_or(RecycleTapeFileConversionError::InvalidTimestamp(
312                val.archive_file_creation_time as i64,
313            ))?;
314
315        let checksums = val
316            .checksum
317            .iter()
318            .map(Checksum::try_from)
319            .collect::<Result<Vec<_>, _>>()
320            .map_err(|e| RecycleTapeFileConversionError::ChecksumConversion(e.to_string()))?;
321
322        Ok(File {
323            archive_file: ArchiveFile {
324                id: val.archive_file_id,
325                storage_class: val.storage_class,
326                creation_time,
327                size: val.size_in_bytes,
328                checksums,
329            },
330            tape_file: TapeFile {
331                vid: val.vid,
332                copy_nb: val.copy_nb,
333                block_id: val.block_id,
334                f_seq: val.fseq,
335            },
336            disk_file: DiskFile {
337                instance: val.disk_instance,
338                id: val.disk_file_id,
339                id_when_deleted: Some(val.disk_file_id_when_deleted),
340                owner: Some(Owner(val.disk_file_uid as u32, val.disk_file_gid as u32)),
341                path: val.disk_file_path,
342            },
343            virtual_org: Some(val.virtual_organization),
344        })
345    }
346}

348impl TryFrom<TapeFileLsItem> for File {
349    type Error = TapeFileLsItemConversionError;
350
351    fn try_from(val: TapeFileLsItem) -> Result<Self, Self::Error> {
352        Ok(File {
353            archive_file: val
354                .af
355                .ok_or_else(|| TapeFileLsItemConversionError("archive_file".into()))?
356                .try_into()
357                .map_err(|e: ArchiveFileConversionError| {
358                    TapeFileLsItemConversionError(format!("archive_file: {}", e))
359                })?,
360            tape_file: val
361                .tf
362                .ok_or_else(|| TapeFileLsItemConversionError("tape_file".into()))?
363                .into(),
364            disk_file: val
365                .df
366                .ok_or_else(|| TapeFileLsItemConversionError("disk_file".into()))?
367                .into(),
368            virtual_org: None,
369        })
370    }

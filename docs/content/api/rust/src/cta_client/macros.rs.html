4//! Helper macros for building admin command protobuf messages.
5//!
6//! The main macro is `admin_cmd!`, which builds an
7//! [`cta_protobuf::cta::admin::AdminCmd`] message with typed parameter setters.
8
9/// Build an [`AdminCmd`] message with typed parameter setters.
10///
11/// The macro mirrors the protobuf structure of a CTA admin command:
12/// `Command.SubCommand { Field: Type => value, ... }`.
13///
14/// # Syntax
15///
16/// ```text
17/// admin_cmd!(Command.SubCommand {
18///     FieldName: Type   => expr,        // required — expr must be the value type
19///     FieldName: Type?  => option_expr, // optional — expr must be `Option<T>`
20/// })
21/// ```
22///
23/// Supported field types:
24/// * `str` — a string value, inserted via [`OptionString`]
25/// * `u64` — an unsigned integer, inserted via [`OptionUInt64`]
26/// * `str_list` — a list of strings, inserted via [`OptionStrList`]
27///
28/// # Example — required fields only
29///
30/// Query a specific tape file by its archive ID:
31///
32/// ```no_run
33/// # use cta_client::admin_cmd;
34/// let cmd = admin_cmd!(Tapefile.SubcmdLs {
35///     ArchiveFileId: u64 => 4294967296u64,
36/// });
37/// ```
38///
39/// [`AdminCmd`]: cta_protobuf::cta::admin::AdminCmd
40/// [`OptionString`]: cta_protobuf::cta::admin::OptionString
41/// [`OptionUInt64`]: cta_protobuf::cta::admin::OptionUInt64
42/// [`OptionStrList`]: cta_protobuf::cta::admin::OptionStrList
43#[macro_export]
44macro_rules! admin_cmd {
45    // Entry point
46    ($cmd:ident . $sub_cmd:ident { $($fields:tt)* }) => {{
47        #[allow(unused_imports, reason = "macro expansion is data-dependent")]
48        use cta_protobuf::cta::admin::{
49            AdminCmd, OptionString, OptionUInt64, OptionStrList,
50            option_string::Key as StringKeys,
51            option_u_int64::Key as U64Keys,
52            option_str_list::Key as StrListKeys,
53            admin_cmd
54        };
55        #[allow(unused_mut, reason = "macro expansion is data-dependent")]
56        let mut cmd = AdminCmd {
57            cmd: admin_cmd::Cmd::$cmd.into(),
58            subcmd: admin_cmd::SubCmd::$sub_cmd.into(),
59            ..Default::default()
60        };
61        admin_cmd!(@fields cmd, $($fields)*);
62        cmd
63    }};
64
65    // base case: no fields left
66    (@fields $cmd:ident, ) => {};
67
68    // optional field: handle `?`
69    (@fields $cmd:ident, $key:ident : $type:ident ? => $val:expr $(, $($rest:tt)*)?) => {
70        if let Some(val) = $val {
71            admin_cmd!(@field $cmd, $type, $key, val);
72        }
73        admin_cmd!(@fields $cmd, $($($rest)*)?);
74    };
75
76    // required field: no `?`
77    (@fields $cmd:ident, $key:ident : $type:ident => $val:expr $(, $($rest:tt)*)?) => {
78        admin_cmd!(@field $cmd, $type, $key, $val);
79        admin_cmd!(@fields $cmd, $($($rest)*)?);
80    };
81
82    // final types
83    (@field $cmd:ident, str, $key:ident, $val:expr) => {
84        $cmd.option_str.push(OptionString {
85            key: StringKeys::$key.into(),
86            value: $val.to_string(),
87        });
88    };
89    (@field $cmd:ident, u64, $key:ident, $val:expr) => {
90        $cmd.option_uint64.push(OptionUInt64 {
91            key: U64Keys::$key.into(),
92            value: $val.into(),
93        });
94    };
95    (@field $cmd:ident, str_list, $key:ident, $val:expr) => {
96        $cmd.option_str_list.push(OptionStrList {
97            key: StrListKeys::$key.into(),
98            item: $val.into_iter().map(|s| s.to_string()).collect(),
99        });
100    };
101}

4//! Errors produced by the EOS client library
5
6use std::num::ParseIntError;
7
8use eos_protobuf::eos::rpc::MdId;
9
10/// An Error coming from the EOS API client
11#[derive(thiserror::Error, Debug)]
12pub enum Error {
13    /// No endpoint is registered for the requested disk instance in the
14    /// [`crate::EosEndpointMap`].
15    #[error("Disk instance '{0}' not found")]
16    DiskInstanceNotFound(String),
17    /// The connection to the EOS endpoint could not be established.
18    #[error("gRPC Connection Error: {0}")]
19    Rpc(cta_grpc_common::Error),
20    /// EOS returned an error status for the call.
21    #[error("gRPC Status Error: {0}")]
22    Tonic(#[from] tonic::Status),
23    /// A numeric identifier could not be parsed from its string form.
24    #[error("Error parsing '{0}': {1}")]
25    ParseInt(String, ParseIntError),
26    /// The disk file id is not usable (EOS reserves the value `0`).
27    #[error("Invalid disk file id: {0}")]
28    InvalidDiskFileId(u64),
29    /// EOS answered successfully but with a payload that violates the
30    /// expectations of the caller (missing metadata, too many results, …).
31    #[error("Unexpected return value: {0}")]
32    UnexpectedReturnValue(String),
33    /// No namespace entry exists for the queried identifier.
34    #[error("Not found: {0:?}")]
35    NotFound(MdId),
36    /// Walking up the parent chain reached the namespace root, so there is no
37    /// parent container left to create.
38    #[error("Root container reached")]
39    RootContainerReached,
40    /// Invalid file/container path
41    #[error("Invalid path: {0}")]
42    InvalidPath(String),
43}

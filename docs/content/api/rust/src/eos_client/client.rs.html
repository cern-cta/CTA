4//! Client library for the EOS namespace gRPC API.
5//!
6//! [`EosGrpcClient`] covers the subset of the EOS RPC service that CTA tools
7//! need: querying file and container metadata, creating containers and
8//! inserting file entries into the namespace.
9//!
10//! An [`EosEndpointMap`] registry maps a *disk instance* name to an
11//! [`EndpointConfig`] and exposes the most common operations directly, opening
12//! the connection on demand.
13//!
14//! ```no_run
15//! # use std::collections::HashMap;
16//! # use eos_client::EosEndpointMap;
17//! # use cta_grpc_common::EndpointConfig;
18//! # async fn example(configs: HashMap<String, EndpointConfig>) -> Result<(), Box<dyn std::error::Error>> {
19//! let mut endpoints = EosEndpointMap::from(configs);
20//! let exists = endpoints
21//!     .check_file_exists_by_disk_id("eosctatape", "1234")
22//!     .await?;
23//! # Ok(())
24//! # }
25//! ```
26

42/// Permission bits applied to namespace entries created by this crate:
43/// `rwx` for the owner, `rw` for group and others
44pub static DEFAULT_FILE_MODE: LazyLock<Mode> = LazyLock::new(|| {
45    (Mode::S_IRWXU | Mode::S_IRGRP | Mode::S_IWGRP | Mode::S_IROTH | Mode::S_IWOTH) &
46        // Filemode: filter out S_ISUID, S_ISGID and S_ISVTX because EOS does not follow POSIX semantics for these bits
47        !(Mode::S_ISUID | Mode::S_ISGID | Mode::S_ISVTX)
48});

50/// This struct encapsulates an EOS GRPC client, abstracting out details such as authentication
51/// and streaming.
52#[derive(Debug)]
53pub struct EosGrpcClient {
54    inner: EosClient<InterceptedService<Channel, AuthorizationInterceptor>>,
55    authentication: JwtAuth,
56}

58impl EosGrpcClient {
59    /// Build a client from a config structure
60    ///
61    /// # Errors
62    ///
63    /// Propagates the connection errors of [`EndpointConfig::build_channel`]
64    /// and the token errors of [`AuthorizationInterceptor::new`].
65    pub async fn new(config: &EndpointConfig) -> Result<Self, RpcError> {
66        let channel = config.build_channel().await?;
67        let client = EosClient::new(InterceptedService::new(
68            channel,
69            AuthorizationInterceptor::new(config.authentication.clone())?,
70        ));
71        Ok(Self {
72            inner: client,
73            authentication: config.authentication.clone(),
74        })
75    }

77    /// Get item metadata
78    ///
79    /// Issues an `md` query and collapses the response stream into the single
80    /// result the caller expects.
81    ///
82    /// # Errors
83    ///
84    /// Returns [`Error::NotFound`] when the stream is empty and
85    /// [`Error::UnexpectedReturnValue`] when it holds more than one item.
86    async fn get_metadata(&mut self, r#type: Type, id: MdId) -> Result<MdResponse, Error> {
87        log::debug!("Retrieving EOS metadata for file {id:?}");
88
89        let response_stream = self
90            .inner
91            .md(with_auth_key_from!(
92                self.authentication,
93                MdRequest {
94                    r#type: r#type.into(),
95                    id: Some(id.clone()),
96                    role: None,
97                    selection: None,
98                }
99            ))
100            .await?
101            .into_inner();
102
103        let res = response_stream.collect::<Result<Vec<_>, _>>().await?;
104
105        match res.len() {
106            0 => Err(Error::NotFound(id)),
107            1 => Ok(res[0].clone()),
108            2.. => Err(Error::UnexpectedReturnValue(format!(
109                "The operation should have returned at most 1 result but returned {}",
110                res.len()
111            ))),
112        }
113    }

115    /// Get all metadata regarding a file
116    ///
117    /// # Errors
118    ///
119    /// Returns [`Error::NotFound`] if the file does not exist
120    pub async fn get_file_metadata(&mut self, id: MdId) -> Result<FileMdProto, Error> {
121        let md_resp = self.get_metadata(Type::File, id.clone()).await?;
122        match md_resp.fmd {
123            Some(mut fmd) => {
124                if fmd.id == 0 {
125                    // Important: EOS responds with id 0 when the item doesn't exist
126                    Err(Error::NotFound(id))
127                } else {
128                    // TODO: Remove this once EOS fixes its behaviour
129                    // see https://its.cern.ch/jira/browse/EOS-6652
130                    #[expect(deprecated)]
131                    if let Some(ref cs) = fmd.checksum
132                        && fmd.checksums.is_empty()
133                    {
134                        fmd.checksums = vec![cs.clone()];
135                    }
136                    Ok(fmd)
137                }
138            }
139            None => Err(Error::UnexpectedReturnValue(
140                "No file metadata was returned".into(),
141            )),
142        }
143    }

145    /// Get all metadata regarding a container
146    ///
147    /// # Errors
148    ///
149    /// Returns [`Error::NotFound`] if the container does not exist.
150    pub async fn get_container_metadata(&mut self, id: MdId) -> Result<ContainerMdProto, Error> {
151        let md_resp = self.get_metadata(Type::Container, id.clone()).await?;
152        match md_resp.cmd {
153            Some(cmd) => {
154                if cmd.id == 0 {
155                    // Important: EOS reponds with id 0 when the item doesn't exist
156                    Err(Error::NotFound(id))
157                } else {
158                    Ok(cmd)
159                }
160            }
161            None => Err(Error::UnexpectedReturnValue(
162                "No container metadata was returned".into(),
163            )),
164        }
165    }

167    /// Add a new container at a path, with a given storage class. Optionally create the parent
168    /// container(s) if they don't exist
169    ///
170    /// The operation is idempotent: if a container already exists at `path`,
171    /// its id is returned unchanged. `file_mode` defaults to 766 ([`DEFAULT_FILE_MODE`]).
172    ///
173    /// # Errors
174    ///
175    /// Returns [`Error::NotFound`] if the parent does not exist and
176    /// `create_parents` is `false`, and [`Error::RootContainerReached`] if
177    /// `path` has no parent at all.
178    pub async fn add_container(
179        &mut self,
180        path: &str,
181        storage_class: &str,
182        create_parents: bool,
183        file_mode: Option<Mode>,
184    ) -> Result<u64, Error> {
185        match self.get_container_disk_id_by_path(path).await {
186            Ok(c_id) => {
187                return {
188                    log::info!(
189                        "Container with path '{}' already exists (id: {})",
190                        path,
191                        c_id
192                    );
193                    Ok(c_id)
194                };
195            }
196            Err(Error::NotFound(_)) => {
197                // there isn't a container with that ID yet, proceed
198            }
199            Err(e) => return Err(e),
200        };
201
202        log::info!("Adding container with path '{path}'");
203
204        let current_path = PathBuf::from(path.strip_suffix("/").unwrap_or(path));
205        let parent = current_path.parent();
206
207        let parent_path = match parent {
208            None => return Err(Error::RootContainerReached),
209            Some(path) => path,
210        }
211        .to_string_lossy()
212        .to_string();
213
214        // the parent container should exist. if not, we'll add it
215        match self.get_container_disk_id_by_path(&parent_path).await {
216            Ok(_) => {
217                // there is already a parent container. do nothing.
218            }
219            Err(Error::NotFound(md_id)) => {
220                // if we're supposed to create the parent(s) container(s), let's go up the chain
221                if create_parents {
222                    log::info!(
223                        "Parent container '{}' doesn't exist yet. Creating it.",
224                        parent_path
225                    );
226                    // we have to pin the future, because of recursion
227                    Box::pin(self.add_container(
228                        &parent_path,
229                        storage_class,
230                        create_parents,
231                        file_mode,
232                    ))
233                    .await?;
234                } else {
235                    // otherwise, fail already
236                    return Err(Error::NotFound(md_id));
237                }
238            }
239            Err(e) => return Err(e),
240        };
241
242        let secs_since_epoch: u64 = system_time_now();
243
244        // now let's create the actual target container
245        let mut dir = ContainerMdProto {
246            path: path.into(),
247            name: current_path
248                .file_name()
249                .ok_or(Error::InvalidPath(path.into()))?
250                .to_string_lossy()
251                .to_string()
252                .into(),
253            mode: file_mode.unwrap_or(*DEFAULT_FILE_MODE).bits(),
254            ctime: Some(Time {
255                sec: secs_since_epoch,
256                n_sec: 0,
257            }),
258            mtime: Some(Time {
259                sec: secs_since_epoch,
260                n_sec: 0,
261            }),
262            ..Default::default()
263        };
264
265        dir.xattrs
266            .insert("sys.archive.storage_class".into(), storage_class.into());
267
268        let res = self
269            .inner
270            .container_insert(with_auth_key_from!(
271                self.authentication,
272                ContainerInsertRequest {
273                    container: vec![dir],
274                    inherit_md: true,
275                }
276            ))
277            .await?;
278
279        let reply = res.into_inner();
280
281        log::debug!("InsertReply: {reply:#?}");
282
283        log::debug!("Confirming container's existence after creation ({path})");
284
285        match self.get_container_disk_id_by_path(path).await {
286            Ok(c_id) => {
287                log::info!(
288                    "Container with id '{}' and path '{}' confirmed to exist",
289                    c_id,
290                    path
291                );
292                Ok(c_id)
293            }
294            Err(e) => Err(e),
295        }
296    }

298    /// Get a file's path from its `disk_file_id`
299    ///
300    /// # Errors
301    ///
302    /// Returns [`Error::ParseInt`] if `disk_file_id` is not a number,
303    /// [`Error::InvalidDiskFileId`] if it is `0` and [`Error::NotFound`] if no
304    /// such file exists.
305    pub async fn get_file_path_by_disk_id(&mut self, disk_file_id: &str) -> Result<String, Error> {
306        let int_id = disk_file_id
307            .parse::<u64>()
308            .map_err(|e| Error::ParseInt(disk_file_id.into(), e))?;
309
310        if int_id == 0 {
311            return Err(Error::InvalidDiskFileId(int_id));
312        }
313
314        let id = MdId {
315            r#type: Type::File.into(),
316            id: int_id,
317            path: "".into(),
318            ino: 0,
319        };
320
321        let md = self.get_file_metadata(id).await?;
322
323        Ok(String::from_utf8_lossy(&md.path.clone()).to_string())
324    }

331    pub async fn get_file_disk_id_by_path(&mut self, file_path: &str) -> Result<u64, Error> {
332        let id = MdId {
333            r#type: Type::File.into(),
334            id: 0,
335            path: file_path.into(),
336            ino: 0,
337        };

339        let md = self.get_file_metadata(id).await?;
340        Ok(md.id)

363    /// Check whether a file exists, given its disk file id
364    ///
365    /// # Errors
366    ///
367    /// Fails for any error other than "not found", e.g. an unparsable
368    /// `disk_file_id` or a failing RPC.
369    pub async fn check_file_exists_by_disk_id(
370        &mut self,
371        disk_file_id: &str,
372    ) -> Result<bool, Error> {
373        match self.get_file_path_by_disk_id(disk_file_id).await {
374            Ok(_) => Ok(true),
375            Err(Error::NotFound(_)) => Ok(false),
376            Err(e) => Err(e),
377        }
378    }

380    /// Get the current ID counters for containers and files, respectively
381    ///
382    /// # Errors
383    ///
384    /// Returns [`Error::Tonic`] if the `ns_stat` call fails.
385    pub async fn get_current_ids(&mut self) -> Result<(u64, u64), Error> {
386        let res = self
387            .inner
388            .ns_stat(with_auth_key_from!(self.authentication, NsStatRequest {}))
389            .await?;
390
391        let res = res.into_inner();
392
393        Ok((res.current_cid, res.current_fid))
394    }

396    /// Insert new files into the EOS namespace
397    ///
398    /// The reply reports the per-file outcome; a successful return value only
399    /// means the RPC itself succeeded.
400    ///
401    /// # Errors
402    ///
403    /// Returns [`Error::Tonic`] if the `file_insert` call fails.
404    pub async fn insert_files(&mut self, files: &[FileMdProto]) -> Result<InsertReply, Error> {
405        let resp = self
406            .inner
407            .file_insert(with_auth_key_from!(
408                self.authentication,
409                FileInsertRequest {
410                    files: files.to_vec()
411                }
412            ))
413            .await?;
414        Ok(resp.into_inner())
415    }

418/// A registry of EOS endpoints, keyed by CTA disk instance name.
419/// Note that each call opens a fresh connection; connections are not pooled.
420pub struct EosEndpointMap {
421    pub(crate) endpoint_map: HashMap<String, EndpointConfig>,
422}
423
424/// A macro to generate proxy methods for the `EosEndpointMap` struct.
425///
426/// Each generated method resolves the disk instance to a client and delegates
427/// to the [`EosGrpcClient`] method of the same name.
428macro_rules! endpoint_method {
429    ($name:ident, ($($param:ident: $type:ty),*) => $ret:ty) => {
430        /// Resolves `disk_instance` to an [`EosGrpcClient`] and forwards the
431        /// call to its equally named method.
432        ///
433        /// # Errors
434        ///
435        /// Returns [`Error::DiskInstanceNotFound`] if the instance is unknown or
436        /// [`Error::Rpc`] if the connection cannot be established. Returns any
437        /// error of the underlying client method.
438        pub async fn $name(
439            &mut self,
440            disk_instance: &str,
441            $($param: $type),*
442        ) -> Result<$ret, Error> {
443            let mut endpoint = self.get_client(disk_instance).await?;
444            endpoint.$name($($param),*).await
445        }
446    };
447}
448
449impl EosEndpointMap {
450    /// Opens a new connection to the endpoint registered for `index`.
451    ///
452    /// # Errors
453    ///
454    /// Returns [`Error::DiskInstanceNotFound`] if `index` is not in the map.
455    /// Returns [`Error::Rpc`] if the instance is registered but the connection
456    /// cannot be established.
457    pub async fn get_client(&mut self, index: impl AsRef<str>) -> Result<EosGrpcClient, Error> {
458        let config = self
459            .endpoint_map
460            .get_mut(index.as_ref())
461            .ok_or_else(|| Error::DiskInstanceNotFound(index.as_ref().to_string()))?;
462        EosGrpcClient::new(config).await.map_err(Error::Rpc)
463    }
464
465    endpoint_method!(get_file_metadata, (id: MdId) => FileMdProto);
466    endpoint_method!(get_file_path_by_disk_id, (disk_file_id: &str) => String);
467    endpoint_method!(get_file_disk_id_by_path, (path: &str) => u64);
468    endpoint_method!(get_current_ids, () => (u64, u64));
469    endpoint_method!(check_file_exists_by_disk_id, (disk_file_id: &str) => bool);
470    endpoint_method!(get_container_metadata, (id: MdId) => ContainerMdProto);
471    endpoint_method!(get_container_disk_id_by_path, (path: &str) => u64);
472    endpoint_method!(add_container, (path: &str, storage_class: &str, create_parents: bool, file_mode: Option<Mode>) => u64);
473    endpoint_method!(insert_files, (files: &[FileMdProto]) => InsertReply);
474}
475
476impl From<HashMap<String, EndpointConfig>> for EosEndpointMap {
477    fn from(endpoint_map: HashMap<String, EndpointConfig>) -> Self {
478        EosEndpointMap { endpoint_map }
479    }
